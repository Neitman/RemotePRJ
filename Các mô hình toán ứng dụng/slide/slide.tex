\begin{frame}{Hai loài sinh vật trọng yếu trong hệ sinh thái Vịnh}
  \begin{columns}[t]
    \begin{column}{0.49\textwidth}
      \begin{block}{\textbf{Cá mòi dầu (\textit{Brevoortia tyrannus})}}
        \footnotesize
        \textbf{\color{OceanNavy}``Loài cá quan trọng nhất trong vịnh Chesapeake''}
        \begin{itemize}
          \item \textbf{Sinh học:} Dài $\sim 45$ cm, bơi thành đàn khổng lồ tại tầng mặt ven bờ; sinh sản quanh năm.
          \item \textbf{Ăn lọc (\textit{Filter-feeder}):} Tiêu thụ tảo phù du, \textbf{chống phú dưỡng hóa}, giữ sạch nước vịnh.
          \item \textbf{Nền tảng chuỗi thức ăn:} Thức ăn của cá vược sọc, cá xanh, cá mập cát, chim ưng biển (\textit{Osprey}).
          \item \textbf{Kinh tế:} Làm bột cá, mồi câu cua xanh, ép dầu giàu axit béo \textbf{Omega-3} (EPA/DHA) giá trị cao.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.49\textwidth}
      \begin{exampleblock}{\textbf{Cá vược sọc (\textit{Morone saxatilis})}}
        \footnotesize
        \textbf{\color{OceanTeal}Kẻ săn mồi đỉnh cao -- Biểu tượng của Vịnh}
        \begin{itemize}
          \item \textbf{Sinh học:} Kích thước lớn ($60 - 90$ cm, $5 - 15$ kg), di cư ngược dòng (\textit{anadromous}) vào sông đẻ trứng.
          \item \textbf{Tập tính săn mồi:} Cá mòi dầu chiếm tỷ trọng áp đảo trong khẩu phần sinh khối và năng lượng trao đổi chất.
          \item \textbf{Nghề cá thương mại:} Cung cấp thịt cá thương phẩm cao cấp cho chuỗi nhà hàng trên toàn quốc.
          \item \textbf{Nghề cá giải trí:} Ngành câu cá thu hút hàng trăm nghìn cần thủ, đóng góp \textbf{hàng tỷ USD vào GDP}.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Khủng hoảng khai thác \& Bi kịch tài nguyên chung}
  \begin{columns}[t]
    \begin{column}{0.50\textwidth}
      \begin{alertblock}{\textbf{Thực trạng suy giảm nghiêm trọng}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Cá mòi dầu:} Sinh khối giảm sút liên tục từ thập niên 1970 do tàu công nghiệp dùng lưới vây khổng lồ (\textit{purse seine}) phối hợp máy bay trinh sát càn quét.
          \item \textbf{Cá vược sọc:} Bị xếp vào diện \textbf{đang bị đánh bắt quá mức} (\textit{overfished}) theo Báo cáo NEFSC (2019).
          \item \textbf{Hệ lụy kép:} Thiếu cá mòi dầu khiến cá vược sọc suy dinh dưỡng, suy giảm miễn dịch và bùng phát dịch bệnh lao cá do \textit{Mycobacteriosis}.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Bi kịch tài nguyên chung (\textit{Hardin, 1968})}}
        \footnotesize
        \begin{itemize}
          \item Ngư trường là tài sản sở hữu chung mở.
          \item Từng tàu cá chạy đua tối đa hóa sản lượng trước mắt, bỏ qua ngoại ứng tiêu cực (\textit{negative externalities}) làm cạn kiệt khả năng tái sinh của nguồn tài nguyên.
        \end{itemize}
      \end{block}

      \vspace{0.1cm}

      \begin{exampleblock}{\textbf{Nghịch lý đánh giá trữ lượng đơn loài}}
        \footnotesize
        Báo cáo SEDAR 69 (2020) cho rằng cá mòi dầu ``chưa bị đánh bắt quá mức'' vì chỉ tính theo mô hình đơn loài! Bỏ qua hoàn toàn nhu cầu thức ăn sống còn của cá vược sọc!
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Hệ phương trình vi phân Lotka--Volterra tổng quát}
  \begin{block}{\textbf{Mô hình Lotka--Volterra mở rộng 2 loài (Panayotova et al., 2023)}}
    Tốc độ biến thiên sinh khối con mồi $x(t)$ và kẻ săn mồi $y(t)$ được mô tả qua hệ phương trình vi phân:
    \begin{equation}
    \label{eq:lv_system}
    \left\{
    \begin{aligned}
    \frac{dx}{dt} &= x(r - ax - by), \\
    \frac{dy}{dt} &= y(-e - cy + dx).
    \end{aligned}
    \right.
    \end{equation}
  \end{block}

  \vspace{0.15cm}

  \begin{columns}[t]
    \begin{column}{0.49\textwidth}
      \begin{block}{\textbf{Khắc phục hạn chế mô hình cổ điển}}
        \footnotesize
        \begin{itemize}
          \item Mô hình cổ điển (không có $-ax^2, -cy^2$) dự báo dao động điều hòa đóng không tắt (kém thực tế).
          \item Số hạng \textbf{tự giới hạn logistic} ($-ax^2$ và $-cy^2$) phản ánh sức chứa hữu hạn của vịnh, tạo ra sự ổn định tiệm cận.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.49\textwidth}
      \begin{exampleblock}{\textbf{Ý nghĩa động lực học riêng lẻ}}
        \footnotesize
        \begin{itemize}
          \item Khi vắng kẻ săn mồi ($y=0$): $\dot{x} = x(r - ax)$, quần thể cá mòi dầu hội tụ về sức chứa $K_x = r/a$.
          \item Khi vắng con mồi ($x=0$): $\dot{y} = y(-e - cy) \le -ey$, quần thể cá vược suy giảm theo hàm mũ về 0.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{5 Giả thiết nền tảng \& Ưu thế của mô hình}
  \begin{columns}[t]
    \begin{column}{0.50\textwidth}
      \begin{block}{\textbf{5 Giả thiết sinh học nền tảng}}
        \footnotesize
        \begin{enumerate}
          \item \textbf{Tăng trưởng logistic của con mồi:} Khi $y=0$, $\dot{x} = rx(1 - x/K_x)$ với $K_x = r/a$.
          \item \textbf{Thức ăn duy nhất:} Cá mòi dầu là nguồn thức ăn chính duy trì đàn cá vược sọc.
          \item \textbf{Suy giảm hàm mũ:} Khi $x=0$, đàn săn mồi suy tàn nhanh về 0.
          \item \textbf{Hệ sinh thái cô lập:} Bỏ qua tương tác với loài thứ yếu và biến động ngẫu nhiên khí hậu.
          \item \textbf{Đồng nhất cấu trúc:} Không phân biệt tuổi, kích thước cá thể và phân bố địa lý.
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{\textbf{Ưu thế vượt trội của mô hình}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Khả thi giải tích (\textit{Analytically tractable}):} Số lượng tham số vừa đủ để giải tích tường minh các điểm cân bằng, Jacobi và tính ổn định.
          \item \textbf{Tái hiện dao động trễ pha (\textit{Phase-shift}):} Mô tả chính xác quy luật dao động bùng nổ -- suy giảm tự nhiên giữa 2 loài.
          \item \textbf{Tránh hiện tượng quá khớp (\textit{Overfitting}):} Quy mô 6 tham số là tối ưu để khớp tin cậy với chuỗi dữ liệu 23 năm thực tế.
        \end{itemize}
      \end{exampleblock}

\begin{frame}{Tính toán Trạng thái cân bằng dừng đồng tồn tại $Q(x^*, y^*)$}
  \begin{block}{\textbf{Thiết lập hệ phương trình cân bằng dừng}}
    \footnotesize
    Tại điểm cân bằng đồng tồn tại tự nhiên ($x^* > 0, y^* > 0$), đạo hàm theo thời gian bằng 0:
    \begin{equation}
    \left\{
    \begin{aligned}
    r - ax^* - by^* &= 0 \\
    -e - cy^* + dx^* &= 0
    \end{aligned}
    \right.
    \iff
    \begin{pmatrix}
    a & b \\
    -d & c
    \end{pmatrix}
    \begin{pmatrix}
    x^* \\ y^*
    \end{pmatrix}
    =
    \begin{pmatrix}
    r \\ -e
    \end{pmatrix}
    \end{equation}
  \end{block}

  \vspace{0.1cm}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{\textbf{Công thức nghiệm tường minh (Định lý Cramer)}}
        \footnotesize
        \begin{align}
        x^* &= \frac{rc + be}{ac + bd} = \frac{2.407705}{17.661755} \approx \mathbf{0.1363} \\
        y^* &= \frac{rd - ae}{ac + bd} = \frac{5.097949}{17.661755} \approx \mathbf{0.2886}
        \end{align}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[colback=OceanLight, colframe=OceanTeal, arc=2mm, title=\textbf{Ý nghĩa thực tế cho Vịnh Chesapeake}]
        \footnotesize
        Quy đổi theo tỷ lệ chuẩn hóa $10^9$ con:
        \begin{itemize}
          \item \textbf{Cá mòi dầu:} $x^* \approx \mathbf{136.3\text{ triệu con}}$.
          \item \textbf{Cá vược sọc:} $y^* \approx \mathbf{288.6\text{ triệu con}}$.
        \end{itemize}
        Đây là trạng thái nền tảng sinh thái bền vững tự nhiên khi chưa chịu tác động đánh bắt thương mại của con người.
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Khái niệm Độ đàn hồi sinh học (Resilience -- Pimm \& Lawton 1978)}
  \begin{columns}[t]
    \begin{column}{0.50\textwidth}
      \begin{block}{\textbf{Định nghĩa theo Pimm \& Lawton (1978)}}
        \footnotesize
        \textbf{Độ đàn hồi sinh học ($R$)} là tốc độ mà một hệ sinh thái tự ổn định và phục hồi trở về trạng thái cân bằng bền vững ban đầu sau một cú sốc hoặc nhiễu loạn môi trường đột ngột.
        \begin{equation}
        \tau = \frac{-1}{\mathrm{Re}(\lambda_m)} \quad \Longleftrightarrow \quad R = \frac{1}{\tau} = -\mathrm{Re}(\lambda_m)
        \end{equation}
        \begin{itemize}
          \item $\lambda_m$: Giá trị riêng chủ đạo (\textit{leading eigenvalue}) có phần thực lớn nhất (gần trục ảo nhất) của ma trận Jacobian.
          \item $\tau$: Thời gian hồi phục (\textit{return time to equilibrium}).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[colback=OceanLight, colframe=OceanNavy, arc=2mm, title=\textbf{Ý nghĩa cơ chế động lực học}]
        \footnotesize
        \begin{itemize}
          \item Độ đàn hồi chỉ có nghĩa khi điểm cân bằng \textbf{ổn định tiệm cận}: $\mathrm{Re}(\lambda_m) < 0$.
          \item \textbf{$R$ và $\tau$ tỷ lệ nghịch:}
          \begin{itemize}
            \item $R$ càng lớn $\implies$ $\tau$ càng nhỏ $\implies$ Hệ phục hồi nhanh, sức chống chịu cao.
            \item $R$ nhỏ $\implies$ $\tau$ lớn $\implies$ Hệ tổn thương lâu, nguy cơ suy thoái kéo dài khi gặp biến cố.
          \end{itemize}
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
% SLIDE 17: MA TRẬN JACOBI & TÍNH ỔN ĐỊNH
% ------------------------------------------------------------------------------
\begin{frame}{Ma trận Jacobian \& Khảo sát tính ổn định tiệm cận}
  \begin{block}{\textbf{Tuyến tính hóa hệ động lực tại điểm cân bằng $Q(x^*, y^*)$}}
    \footnotesize
    Ma trận Jacobi tổng quát của hệ:
    $J(x, y) = \begin{pmatrix} r - 2ax - by & -bx \\ dy & -e - 2cy + dx \end{pmatrix}$.
    Tại điểm dừng $Q(x^*, y^*)$:
    \begin{equation}
    A = J(x^*, y^*) = \begin{pmatrix} -ax^* & -bx^* \\ dy^* & -cy^* \end{pmatrix}
    \end{equation}
  \end{block}

  \vspace{0.1cm}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{\textbf{Vết và Định thức của ma trận $A$}}
        \footnotesize
        \begin{align}
        \mathrm{tr}(A) &= -(ax^* + cy^*) < 0 \\
        \det(A) &= (ac + bd)x^*y^* > 0
        \end{align}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[colback=OceanLight, colframe=OceanTeal, arc=2mm, title=\textbf{Tiêu chuẩn Routh--Hurwitz}]
        \footnotesize
        Do $\mathrm{tr}(A) < 0$ và $\det(A) > 0$, mọi giá trị riêng của $A$ đều có \textbf{phần thực âm} $\implies$ Trạng thái cân bằng đồng tồn tại $Q$ là \textbf{ổn định tiệm cận cục bộ} (\textit{locally asymptotically stable}).
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 18: GIÁ TRỊ RIÊNG & THỜI GIAN PHỤC HỒI

\begin{frame}{Tính toán giá trị riêng \& Thời gian hồi phục của Vịnh Chesapeake}
  \begin{block}{\textbf{Thay thế số liệu cụ thể vào phương trình đặc trưng}}
    \footnotesize
    Phương trình đặc trưng: $\det(\lambda I - A) = \lambda^2 - \mathrm{tr}(A)\lambda + \det(A) = 0$.
    \begin{itemize}
      \item $\mathrm{tr}(A) = -[(0.026)(0.1363) + (0.520)(0.2886)] = \mathbf{-0.1536158}$
      \item $\det(A) = (17.661755)(0.1363)(0.2886) \approx \mathbf{0.694747}$
      \item Biệt thức: $\Delta = (\mathrm{tr}(A))^2 - 4\det(A) = (-0.1536)^2 - 4(0.6947) = \mathbf{-2.75539} < 0$
    \end{itemize}
  \end{block}

  \vspace{0.1cm}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{\textbf{Cặp giá trị riêng phức liên hợp}}

        \end{equation}
        Trùng khớp chính xác với công thức (2) trong bài báo gốc của Panayotova et al. (2023).
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[colback=OceanLight, colframe=OceanNavy, arc=2mm, title=\textbf{Độ đàn hồi \& Thời gian hồi phục}]

  \end{columns}
\end{frame}

\begin{frame}{Mở rộng Bionomic: So sánh 3 Chiến lược Khai thác}
  \begin{block}{\textbf{Hệ phương trình kinh tế sinh thái (Bionomic Model)}}
    \footnotesize
    Tích hợp đánh bắt CPUE $h_i = q_i E_i u_i$ và hàm tô kinh tế $\pi(E_1, E_2) = (p_1 q_1 x - c_1)E_1 + (p_2 q_2 y - c_2)E_2$.
  \end{block}

  \vspace{0.1cm}

  \begin{columns}[t]
    \begin{column}{0.32\textwidth}
      \begin{tcolorbox}[colback=white, colframe=OceanNavy, arc=1.5mm, title=\textbf{1. Chính sách MSY}]
        \scriptsize
        \textbf{Sản lượng tối đa}\\
        $\max Y(E) = \sum q_i E_i u_i^*$\\
        \textbf{Cảnh báo:} MSY đơn loài cá mòi dầu ($E_{Y_m} \approx 1.02$) khiến cá vược \textbf{tuyệt chủng}!\\
        $\blacktriangleright$ \textbf{Khuyến nghị:} Cắt giảm 50\% hạn ngạch ($E_{\mathrm{MSY}} = 0.5$).
      \end{tcolorbox}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{tcolorbox}[colback=white, colframe=OceanBlue, arc=1.5mm, title=\textbf{2. Chính sách MEY}]
        \scriptsize
        \textbf{Lợi nhuận kinh tế tối đa}\\
        $\max \pi(E) = R(E) - C(E)$\\
        Tính chất: $E_{\mathrm{MEY}} < E_{\mathrm{MSY}}$ khi chỉ đánh bắt cá vược.\\
        Thân thiện với sinh thái nhưng khó áp dụng đơn lẻ nếu giá và chi phí không cân xứng.
      \end{tcolorbox}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{tcolorbox}[colback=white, colframe=OceanTeal, arc=1.5mm, title=\textbf{3. Chính sách RMY}]
        \scriptsize
        \textbf{Tối đa hóa độ đàn hồi}\\
        $\max R(E) = -\mathrm{Re}(\lambda_m)$\\
        Đánh bắt cá mòi làm $R$ \textbf{giảm đơn điệu} theo $E_1$. Bất kỳ đánh bắt cá mòi nào cũng làm suy giảm khả năng phục hồi của hệ thống!
      \end{tcolorbox}
    \end{column}

\end{frame}
